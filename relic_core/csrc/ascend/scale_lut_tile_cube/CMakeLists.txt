cmake_minimum_required(VERSION 3.16)
project(relic_scale_lut_tile_cube LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(RUN_MODE npu CACHE STRING "Private hardware build")
set(SOC_VERSION Ascend910B3 CACHE STRING "Explicit verified SoC")
if(NOT DEFINED ASCEND_CANN_PACKAGE_PATH)
    message(FATAL_ERROR "Pass the existing CANN root explicitly; do not install or modify CANN")
endif()
include(${ASCEND_CANN_PACKAGE_PATH}/aarch64-linux/tikcpp/ascendc_kernel_cmake/ascendc.cmake)

# Complete output rows per native Cube tile. N32 and N64 were measured separately on Ascend910B3;
# each uses its own declared Cube-order layout and exact-hash library artifact.
set(RELIC_LUT_OUTPUT_ROWS 32 CACHE STRING "Complete output rows per native Cube tile")
if(NOT RELIC_LUT_OUTPUT_ROWS MATCHES "^(32|64)$")
    message(FATAL_ERROR "RELIC_LUT_OUTPUT_ROWS must be an explicit bounded 32 or 64")
endif()

# K128 is fixed by the prepared table format: the code LUT, the measured 117..127 scale window and
# the pair-repeat order are all defined for four original K32 blocks per tile. A different grain
# would be a different table format, so it is rejected rather than offered as a switch.
set(RELIC_LUT_GROUP_K 4 CACHE STRING "Original K32 blocks per tile; the table format fixes this")
if(NOT RELIC_LUT_GROUP_K STREQUAL "4")
    message(FATAL_ERROR "RELIC_LUT_GROUP_K is fixed at 4 by the prepared table format")
endif()

# One generated translation unit embeds the literal geometry for both the host query functions and
# the device compilation, so a host cannot report N32 while the kernel was built for N64. CANN's
# ascendc_compile_definitions applies to the device compilation only, which is why the geometry is
# written into the source instead of passed as a definition.
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/scale_lut_tile_cube_entry.cpp
"// Generated by CMake; do not edit. Embeds one literal geometry for host and device.\n"
"#define RELIC_LUT_OUTPUT_ROWS ${RELIC_LUT_OUTPUT_ROWS}\n"
"#include \"${CMAKE_CURRENT_SOURCE_DIR}/scale_lut_tile_cube.cpp\"\n")

ascendc_library(relic_scale_lut_tile_cube SHARED
    ${CMAKE_CURRENT_BINARY_DIR}/scale_lut_tile_cube_entry.cpp)
